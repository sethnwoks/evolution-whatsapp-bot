# PSEUDOCODE FOR: create_instance.py
# PURPOSE: Create WhatsApp instance with webhook using Python instead of curl

# ============================================================================
# STEP 1: IMPORT NECESSARY LIBRARIES
# ============================================================================

IMPORT requests library
  ↳ WHY: We need to make HTTP calls (like POST, GET) to Evolution API
  ↳ WHAT IT DOES: Provides functions like requests.post(), requests.get()
  ↳ ALTERNATIVE: Could use 'urllib' (built-in) but requests is cleaner

IMPORT json library  
  ↳ WHY: To convert Python dictionaries to JSON strings and vice versa
  ↳ WHAT IT DOES: json.dumps() = dict → string, json.loads() = string → dict
  ↳ WHEN NEEDED: Formatting output, debugging responses


# ============================================================================
# STEP 2: DEFINE CONFIGURATION CONSTANTS
# ============================================================================

SET API_KEY = "B4sR983jKp2mN7xQ"
  ↳ WHAT: Your authentication token
  ↳ WHY: Evolution API checks this to verify you're authorized
  ↳ WHERE IT CAME FROM: Generated when you set up Evolution, stored in .env
  ↳ SECURITY: In production, load from environment variables, not hardcode

SET BASE_URL = "http://localhost:8080"
  ↳ WHAT: The address where Evolution API is running
  ↳ WHY: So we can build full URLs like "http://localhost:8080/instance/create"
  ↳ COMPONENTS: 
      • http:// = protocol
      • localhost = your computer
      • 8080 = port Evolution is listening on


# ============================================================================
# STEP 3: DEFINE THE MAIN FUNCTION
# ============================================================================

FUNCTION create_instance_with_webhook(instance_name):
  ↳ WHAT: A reusable function that creates instances
  ↳ INPUT: instance_name (string) - e.g., "fire", "bot1", "production"
  ↳ OUTPUT: Python dictionary containing Evolution's response
  ↳ WHY FUNCTION: Reusable - call it multiple times with different names


  # --------------------------------------------------------------------------
  # STEP 3A: BUILD THE REQUEST URL
  # --------------------------------------------------------------------------
  
  SET url = BASE_URL + "/instance/create"
    ↳ RESULT: "http://localhost:8080/instance/create"
    ↳ WHY: This is Evolution's endpoint for creating instances
    ↳ HOW WE KNEW: From Evolution API documentation
    ↳ f-string version: f"{BASE_URL}/instance/create"


  # --------------------------------------------------------------------------
  # STEP 3B: PREPARE HTTP HEADERS
  # --------------------------------------------------------------------------
  
  CREATE headers dictionary with:
    ↳ WHAT: Metadata that goes with the HTTP request
    ↳ FORMAT: Key-value pairs
    
    KEY: "Content-Type", VALUE: "application/json"
      ↳ WHAT: Tells Evolution "I'm sending JSON data, not XML or plain text"
      ↳ WHY: Server needs to know how to parse the body
      ↳ ALTERNATIVE: "application/x-www-form-urlencoded" for form data
    
    KEY: "apikey", VALUE: API_KEY (the variable from step 2)
      ↳ WHAT: Your authentication token
      ↳ WHY: Proves you're allowed to create instances
      ↳ SECURITY: This is sent with EVERY request


  # --------------------------------------------------------------------------
  # STEP 3C: BUILD THE REQUEST BODY (THE DATA WE'RE SENDING)
  # --------------------------------------------------------------------------
  
  CREATE data dictionary with:
    ↳ WHAT: The actual information Evolution needs to create the instance
    ↳ FORMAT: Python dictionary that will become JSON
    
    KEY: "instanceName", VALUE: instance_name (function parameter)
      ↳ WHAT: Name to identify this WhatsApp connection
      ↳ EXAMPLE: "fire", "customer-bot", "sales-automation"
      ↳ RULES: Must be unique, no spaces
    
    KEY: "qrcode", VALUE: True (boolean)
      ↳ WHAT: Whether to generate a QR code for connection
      ↳ TRUE: Generate QR code (scan with phone)
      ↳ FALSE: Use pairing code instead (8-digit number)
    
    KEY: "webhook", VALUE: another dictionary (nested configuration)
      ↳ WHAT: Configuration for where to send incoming message data
      ↳ WHY NESTED: webhook has its own settings
      
      INSIDE webhook dictionary:
      
        KEY: "enabled", VALUE: True (boolean)
          ↳ WHAT: Turn webhooks on or off
          ↳ TRUE: Send webhook data when events happen
          ↳ FALSE: Don't send any webhooks
        
        KEY: "url", VALUE: "http://host.docker.internal:5000/webhook"
          ↳ WHAT: Where Evolution should send webhook data
          ↳ WHY THIS ADDRESS:
              • Evolution is inside Docker container
              • FastAPI runs on your laptop at port 5000
              • "host.docker.internal" = Docker's way to reach host machine
              • "/webhook" = your FastAPI endpoint
          ↳ PRODUCTION: Would be "https://yourserver.com/webhook"
        
        KEY: "byEvents", VALUE: True (boolean)
          ↳ WHAT: Send only specific events (below) vs everything
          ↳ TRUE: Only send events in the "events" array
          ↳ FALSE: Send ALL events (connection, qr, typing, everything)
        
        KEY: "base64", VALUE: False (boolean)
          ↳ WHAT: Encode media (images, videos) as base64 strings
          ↳ TRUE: Images become long text strings in webhook
          ↳ FALSE: Images sent as URLs (lighter, faster)
        
        KEY: "events", VALUE: array containing "MESSAGES_UPSERT"
          ↳ WHAT: List of events to listen for
          ↳ MESSAGES_UPSERT: When new messages arrive
          ↳ OTHER OPTIONS:
              • "MESSAGES_UPDATE" - message edited/deleted
              • "QRCODE_UPDATED" - new QR code generated
              • "CONNECTION_UPDATE" - connection status changed
          ↳ FORMAT: Array/list of strings


  # --------------------------------------------------------------------------
  # STEP 3D: MAKE THE HTTP REQUEST
  # --------------------------------------------------------------------------
  
  CALL requests.post() with parameters:
    ↳ WHAT: Sends HTTP POST request to Evolution API
    ↳ WHY POST: We're creating/sending data (not just reading)
    
    PARAMETER 1: url (where to send)
      ↳ VALUE: "http://localhost:8080/instance/create"
    
    PARAMETER 2: headers=headers (metadata)
      ↳ VALUE: The headers dictionary from step 3B
      ↳ CONTAINS: API key and content type
    
    PARAMETER 3: json=data (the body/payload)
      ↳ VALUE: The data dictionary from step 3C
      ↳ WHAT IT DOES: requests automatically:
          1. Converts Python dict → JSON string
          2. Adds it to request body
          3. Sets Content-Type header (we already did this)
  
  STORE result in variable called "response"
    ↳ WHAT: Object containing Evolution's reply
    ↳ CONTAINS:
        • response.status_code (200, 400, 500, etc.)
        • response.text (raw response as string)
        • response.json() (parsed response as dict)


  # --------------------------------------------------------------------------
  # STEP 3E: PARSE THE RESPONSE
  # --------------------------------------------------------------------------
  
  CALL response.json() and store in "result" variable
    ↳ WHAT: Converts Evolution's JSON response → Python dictionary
    ↳ WHY: So we can access fields like result["status"]
    ↳ EXAMPLE RESPONSE:
        {
          "status": "SUCCESS",
          "qrcode": "data:image/png;base64,iVBORw...",
          "instance": {
            "instanceName": "fire",
            "status": "open"
          }
        }


  # --------------------------------------------------------------------------
  # STEP 3F: DISPLAY THE RESULT
  # --------------------------------------------------------------------------
  
  CALL json.dumps() on result with indent=2
    ↳ WHAT: Converts Python dict → formatted JSON string
    ↳ indent=2: Adds 2-space indentation for readability
    ↳ EXAMPLE OUTPUT:
        {
          "status": "SUCCESS",
          "instance": {
            "name": "fire"
          }
        }
    ↳ WITHOUT indent: Everything on one line (hard to read)
  
  PRINT the formatted JSON
    ↳ WHY: So you can see what Evolution responded with
  
  RETURN result
    ↳ WHY: So if you call this function from other code, you get the response
    ↳ EXAMPLE USAGE:
        response = create_instance_with_webhook("fire")
        if response["status"] == "SUCCESS":
            print("It worked!")


# ============================================================================
# STEP 4: MAIN EXECUTION BLOCK
# ============================================================================

IF this script is run directly (not imported):
  ↳ WHAT: if __name__ == "__main__":
  ↳ WHY: Allows file to be:
      • Run directly: python create_instance.py (executes this block)
      • Imported: from create_instance import ... (skips this block)
  
  PRINT "Creating instance 'fire' with webhook configured..."
    ↳ WHY: User feedback - let them know it started
  
  CALL create_instance_with_webhook("fire")
    ↳ WHAT: Execute the function from step 3
    ↳ PARAMETER: "fire" as the instance name
    ↳ RESULT: Creates instance, prints response
  
  PRINT "✅ Instance created! Check the manager UI or terminal for QR code."
    ↳ WHY: Success message with next steps
